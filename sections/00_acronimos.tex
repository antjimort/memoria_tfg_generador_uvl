\chapter*{Lista de acrónimos}
\markboth{Lista de acrónimos}{}
\addcontentsline{toc}{chapter}{Lista de acrónimos}

\vspace{1em}

\begin{description}[
    labelwidth=2.5cm,
    leftmargin=3cm,
    style=nextline,
    itemsep=0.3em
  ]
  \item[\textbf{API}] Interfaz de programación de aplicaciones
  (\textit{Application Programming Interface})
  \item[\textbf{AST}] Árbol de sintaxis abstracta
  (\textit{Abstract Syntax Tree})
  \item[\textbf{BDD}] Diagramas de decisión binaria
  (\textit{Binary Decision Diagrams})
  \item[\textbf{CapEx}] Gastos de capital
  (\textit{Capital Expenditures})
  \item[\textbf{CI/CD}] Integración continua y entrega o despliegue continuo
  (\textit{Continuous Integration / Continuous Delivery or Deployment})
  \item[\textbf{CLI}] Interfaz de línea de comandos
  (\textit{Command-Line Interface})
  \item[\textbf{CNF}] Forma normal conjuntiva
  (\textit{Conjunctive Normal Form})
  \item[\textbf{CU}] Caso de uso
  \item[\textbf{DOI}] Identificador de objeto digital
  (\textit{Digital Object Identifier})
  \item[\textbf{ECTS}] Sistema Europeo de Transferencia y Acumulación de Créditos (\textit{European Credit Transfer and Accumulation System})
  \item[\textbf{EDT}] Estructura de descomposición del trabajo
  \item[\textbf{E2E}] Extremo a extremo (\textit{End-to-End})
  \item[\textbf{JSON}] \textit{JavaScript Object Notation}
  \item[\textbf{LLM}] Gran Modelo de Lenguaje (\textit{Large Language Model})
  \item[\textbf{OpEx}] Gastos operativos
  (\textit{Operational Expenditures})
  \item[\textbf{PMI}] Instituto de Gestión de Proyectos (\textit{Project Management Institute})
  \item[\textbf{REST}] \textit{Representational State Transfer}
  \item[\textbf{RF}] Requisito funcional
  \item[\textbf{RN}] Requisito de negocio
  \item[\textbf{RNF}] Requisito no funcional
  \item[\textbf{SAT}] Problema de satisfacibilidad booleana
  (\textit{Boolean Satisfiability Problem})
  \item[\textbf{SMT}] Satisfacibilidad módulo teorías
  (\textit{Satisfiability Modulo Theories})

  \item[\textbf{SPL}] \textit{Software Product Lines}
  \item[\textbf{TASOVA}] Red en nuevas Tendencias en Arquitectura Software y Variabilidad
  \item[\textbf{UVL}] \textit{Universal Variability Language}
  \item[\textbf{WASM}] \textit{WebAssembly}
\end{description}
